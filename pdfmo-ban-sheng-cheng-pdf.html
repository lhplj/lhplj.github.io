PDF模板生成PDF

lhplj

2种方式：
1、LibreOffice 可以制作带表单的odt模板文件，再通过 itext7 渲染生成（获取表单字段信息进行数据填充），适合格式固定（表格行数固定），动态的话就不好处理
2、lottemplate 支持类似 freemarker 的脚本来动态生成，适合格式不固定的。但提供的http服务，需要调用接口来生成。

# 使用 lotemplate 通过 docker 启动
https://github.com/Probesys/lotemplate/tree/v2.2.6

# 获取代码
git clone https://github.com/Probesys/lotemplate.git

# 修改 .env 中的密码
SECRET_KEY=blulanic

# 启动服务：服务启动后，就可以通过 http://localhost:8000 来访问它的 API
cd lotemplate
docker-compose up -d

# 创建目录
curl -X PUT -H 'secretkey: blulanic' -H 'directory: blulanic' http://localhost:8000/

# 制作模板：LibreOffice 制作的odt文件模板(/Users/lihuipeng/Downloads/LOTemplate.odt)，内容如下
[html]
<table>
  <tr><th>First Name</th><th>Last Name</th></tr>
  [for $people]
  <tr>
    <td>[foritem firstName]</td>
    <td>[foritem lastName]</td>
  </tr>
  [endfor]
</table>
[endhtml]

# 上传模板
curl -X PUT -H 'secretkey: blulanic' -F file=@/Users/lihuipeng/Downloads/LOTemplate.odt http://localhost:8000/blulanic

{"file":"LOTemplate_2.odt","message":"Successfully uploaded","variables":{"people":{"type":"array","value":[]}}}

# 使用模板生成pdf
curl -X POST \
  -H 'secretkey: blulanic' \
  -H 'Content-Type: application/json' \
  -d '{"name":"output.pdf","variables":{"people":{"type":"array","value":[{"firstName":"san","lastName":"zhang"}]}}}' \
  http://localhost:8000/blulanic/LOTemplate.odt --output test.pdf

# 在 lotemplate 中创建目录 fonts
# 将压缩包中的 otf 文件统一放在 lotemplate/fonts 目录下
# 修改 docker-compose.yml 挂载 字体（在 volumes 下增加 - ./fonts:/usr/share/fonts/custom:ro   # 挂载字体目录，只读）
---
services:
  lo_api:
    image: probesys38/lotemplate:v2.2.6
    # build:
    #  context: .
    #  dockerfile: ./Dockerfile
    #  target: prod
    volumes:
      - ./:/app/
      - ./fonts:/usr/share/fonts/custom:ro   # 挂载字体目录，只读
    ports:
      - "8000:8000"
    environment:
      - SECRET_KEY=$SECRET_KEY
      - NB_WORKERS=$NB_WORKERS
      - MAXTIME=$MAXTIME
      - DISABLE_HTML_HACK=${DISABLE_HTML_HACK:-}
      - LOG_LEVEL=${LOG_LEVEL:-}
    command: "gunicorn -b 0.0.0.0:8000  --access-logfile '-'  --access-logformat '%(h)s %(l)s %(u)s %(t)s  \"%(r)s\" %(s)s %(b)s \"%(f)s\" \"%(a)s\" %(M)s' app:app"

# 重启服务
docker-compose down
docker-compose up -d

java示例

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.nio.charset.StandardCharsets;
import java.nio.file.Paths;
import java.util.Collections;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

import org.springframework.core.io.FileSystemResource;
import org.springframework.http.HttpEntity;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.http.HttpStatus;
import org.springframework.http.MediaType;
import org.springframework.http.ResponseEntity;
import org.springframework.http.client.ClientHttpResponse;
import org.springframework.util.LinkedMultiValueMap;
import org.springframework.util.MultiValueMap;
import org.springframework.web.client.ResponseExtractor;
import org.springframework.web.client.RestClientException;
import org.springframework.web.client.RestTemplate;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.google.common.io.Files;

import lombok.extern.slf4j.Slf4j;

@Slf4j
public class DocumentService {

    private static final RestTemplate restTemplate = new RestTemplate();
    private static final String LOTEMPLATE_URL = "http://localhost:8000";
    private static final String SECRET_KEY = "blulanic"; // 从配置文件读取

    public static void main(String[] args) throws IOException {
//        String templatePath = "/Users/lihuipeng/Downloads/LOTemplate.odt";
//        uploadTemplate(templatePath, "blulanic");
        ObjectMapper mapper = new ObjectMapper();

        // 构建内部 person Map
        Map<String, String> person = new HashMap<>();
        person.put("firstName", "訊息發送者");
        person.put("lastName", "參考編號");

        // 构建 people 列表
        List<Map<String, String>> peopleList = Collections.singletonList(person);

        // 构建包装对象，包含 type 和 value
        Map<String, Object> peopleWrapper = new HashMap<>();
        peopleWrapper.put("type", "array");
        peopleWrapper.put("value", peopleList);

        // 构建根对象，变量名为 people
        Map<String, Object> root = new HashMap<>();
        root.put("people", peopleWrapper);

        // 序列化
        String json = mapper.writeValueAsString(root);
        System.out.println(json);

        byte[] bytes = generateDocument("LOTemplate.odt", SECRET_KEY, json);
        System.out.println(bytes.length);
        Files.write(bytes, Paths.get("/Users/lihuipeng/Downloads/genByLOTemplate.pdf").toFile());
    }

    /**
     * 上传模板文件到 LOTemplate
     */
    public static void uploadTemplate(String templatePath, String directory) {
        String url = LOTEMPLATE_URL + "/" + directory;
        
        HttpHeaders headers = new HttpHeaders();
        headers.setContentType(MediaType.MULTIPART_FORM_DATA);
        headers.set("secretkey", SECRET_KEY);

        MultiValueMap<String, Object> body = new LinkedMultiValueMap<>();
        body.add("file", new FileSystemResource(templatePath));

        HttpEntity<MultiValueMap<String, Object>> requestEntity = 
                new HttpEntity<>(body, headers);

        try {
            ResponseEntity<String> response = restTemplate.exchange(url, HttpMethod.PUT, requestEntity, String.class);

            // 处理响应
            if (response.getStatusCode().is2xxSuccessful()) {
                log.info("模板上传成功，状态码: {}, 响应体: {}", response.getStatusCode(), response.getBody());
                // 可以解析响应体获取更多信息
            } else {
                log.error("模板上传失败，状态码: {}, 响应体: {}", response.getStatusCode(), response.getBody());
                throw new RuntimeException("模板上传失败: " + response.getBody());
            }
        } catch (RestClientException e) {
            log.error("模板上传请求异常", e);
            throw new RuntimeException("上传模板时发生异常", e);
        }
    }

    /**
     * 使用模板和数据生成文档
     */
    public static byte[] generateDocument(String templateName, String directory, String jsonData) {
        String url = LOTEMPLATE_URL + "/" + directory + "/" + templateName;
        String requestBody = String.format("{\"name\": \"output.pdf\", \"variables\": %s}", jsonData);

        // 使用 execute 方法，手动处理请求和响应
        return restTemplate.execute(url, HttpMethod.POST, request -> {
            request.getHeaders().set("secretkey", SECRET_KEY);
            request.getHeaders().setContentType(MediaType.APPLICATION_JSON);
            request.getBody().write(requestBody.getBytes(StandardCharsets.UTF_8));
        }, new ResponseExtractor<byte[]>() {
            @Override
            public byte[] extractData(ClientHttpResponse response) throws IOException {
                HttpStatus statusCode = response.getStatusCode();
                if (statusCode.is2xxSuccessful()) {
                    // 读取输入流到字节数组（JDK 8 兼容方式）
                    try (InputStream is = response.getBody()) {
                        ByteArrayOutputStream baos = new ByteArrayOutputStream();
                        byte[] buffer = new byte[4096];
                        int len;
                        while ((len = is.read(buffer)) != -1) {
                            baos.write(buffer, 0, len);
                        }
                        return baos.toByteArray();
                    }
                } else {
                    // 读取错误信息
                    try (InputStream is = response.getBody()) {
                        ByteArrayOutputStream baos = new ByteArrayOutputStream();
                        byte[] buffer = new byte[4096];
                        int len;
                        while ((len = is.read(buffer)) != -1) {
                            baos.write(buffer, 0, len);
                        }
                        String error = new String(baos.toByteArray(), StandardCharsets.UTF_8);
                        throw new RuntimeException("生成文档失败，状态码: " + statusCode + ", 错误: " + error);
                    }
                }
            }
        });
    }
}
