自定义锁（CAS自旋锁）

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicReference;

public class SpinLockTest {

    // 存储持有锁的线程
    private AtomicReference<Thread> ar = new AtomicReference<>();

    /**
     * 获取锁:获取锁前，ar 中的值必须是 null，即没有线程持有锁，再把当前线程设置为 ar 的值
     */
    public void lock() {
        Thread thread = Thread.currentThread();
        while (!ar.compareAndSet(null, thread)) {
        }
        System.out.println(time() + thread + " get lock");
    }

    /**
     * 释放锁：释放锁前，ar 中的值必须是当前线程，即是当前线程持有锁，才能释放，更新 ar 的值为 null
     */
    public void unlock() {
        Thread thread = Thread.currentThread();
        while (!ar.compareAndSet(thread, null)) {
        }
        System.out.println(time() + thread + " release lock");
    }

    private String time() {
        return LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss.SSS")) + ": ";
    }

    public static void main(String[] args) {
        SpinLockTest sl = new SpinLockTest();

        new Thread(() -> {
            try {
                sl.lock();
                TimeUnit.SECONDS.sleep(2);
            } catch (InterruptedException e) {
                e.printStackTrace();
            } finally {
                sl.unlock();
            }
        }, "t1").start();

        try {
            TimeUnit.MILLISECONDS.sleep(500);
        } catch (InterruptedException e) {
            e.printStackTrace();
        }

        new Thread(() -> {
            try {
                sl.lock();
                TimeUnit.SECONDS.sleep(2);
            } catch (InterruptedException e) {
                e.printStackTrace();
            } finally {
                sl.unlock();
            }
        }, "t2").start();
    }

}
boolean java.util.concurrent.atomic.AtomicReference.compareAndSet(V expect, V update)

Atomically sets the value to the given updated value if the current value ==the expected value.

Parameters:
expect the expected value
update the new value
Returns:
trueif successful. False return indicates that the actual value was not equal to the expected value.

缺点：
1、不支持可重入（第一次lock后，unlock前再lock会失败，可再增加AtomicInteger以标记重入锁的次数，需要调整 lock 及 unlock 逻辑）
2、会占用CPU让CPU空转，短时间的线程任务没问题
3、存在ABA问题（比较的值A中间被改为B又改回A，部分场景没影响；AtomicStampedReference可解决此类问题）
